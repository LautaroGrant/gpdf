cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" GPDF_VERSION LIMIT_COUNT 1)
project(GPDF VERSION ${GPDF_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets PrintSupport Network)
set(PDFIUM_ROOT "${CMAKE_SOURCE_DIR}/third_party/pdfium" CACHE PATH "PDFium binary distribution")
find_path(PDFIUM_INCLUDE_DIR fpdfview.h PATHS "${PDFIUM_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(PDFIUM_LIBRARY NAMES pdfium pdfium.dll PATHS "${PDFIUM_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(pdfium SHARED IMPORTED)
set_target_properties(pdfium PROPERTIES IMPORTED_IMPLIB "${PDFIUM_LIBRARY}" IMPORTED_LOCATION "${PDFIUM_ROOT}/bin/pdfium.dll" INTERFACE_INCLUDE_DIRECTORIES "${PDFIUM_INCLUDE_DIR}")
add_library(pdf_core src/PdfDocument.cpp)
target_link_libraries(pdf_core PUBLIC Qt6::Widgets pdfium)
target_include_directories(pdf_core PUBLIC src)
add_library(gpdf_ui src/Translations.cpp src/AnnotationExport.cpp src/SessionStore.cpp src/NativeScrollCursor.cpp src/CommentPopup.cpp src/UpdateManager.cpp)
target_link_libraries(gpdf_ui PUBLIC pdf_core Qt6::PrintSupport Qt6::Network)
target_include_directories(gpdf_ui PUBLIC src)
target_compile_definitions(gpdf_ui PUBLIC GPDF_VERSION="${PROJECT_VERSION}")
add_executable(GPDF WIN32 src/main.cpp resources/translations.qrc resources/icons.qrc)
target_link_libraries(GPDF PRIVATE gpdf_ui Qt6::PrintSupport Qt6::Network)
add_executable(pdf_smoke tests/smoke.cpp)
target_link_libraries(pdf_smoke PRIVATE pdf_core)
add_executable(ui_smoke tests/ui_smoke.cpp resources/translations.qrc)
target_link_libraries(ui_smoke PRIVATE gpdf_ui Qt6::PrintSupport Qt6::Network)
add_executable(language_smoke tests/language_smoke.cpp resources/translations.qrc)
target_link_libraries(language_smoke PRIVATE gpdf_ui)
target_compile_definitions(language_smoke PRIVATE GPDF_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  enable_language(RC)
  if(MINGW)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/gpdf-icons.o"
      COMMAND "${CMAKE_RC_COMPILER}" --preprocessor=gcc.exe --preprocessor-arg=-E --preprocessor-arg=-xc --preprocessor-arg=-DRC_INVOKED -O coff -i gpdf.rc -o "${CMAKE_CURRENT_BINARY_DIR}/gpdf-icons.o"
      WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources"
      DEPENDS resources/gpdf.rc resources/icons/gpdf.ico resources/icons/pdf-document.ico VERBATIM)
    target_sources(GPDF PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gpdf-icons.o")
  else()
    target_sources(GPDF PRIVATE resources/gpdf.rc)
    set_source_files_properties(resources/gpdf.rc PROPERTIES COMPILE_OPTIONS "/I${CMAKE_CURRENT_SOURCE_DIR}/resources")
  endif()
  target_link_libraries(gpdf_ui PUBLIC user32 gdi32)
  target_link_libraries(GPDF PRIVATE user32 shell32)
  target_link_libraries(ui_smoke PRIVATE user32)
endif()
add_executable(corpus_smoke tests/corpus_smoke.cpp)
target_link_libraries(corpus_smoke PRIVATE pdf_core)
add_executable(render_quality tests/render_quality.cpp resources/translations.qrc)
add_executable(update_smoke tests/update_smoke.cpp)
target_link_libraries(update_smoke PRIVATE gpdf_ui)
target_link_libraries(render_quality PRIVATE gpdf_ui)
foreach(target GPDF pdf_smoke ui_smoke corpus_smoke language_smoke render_quality)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:pdfium>" "$<TARGET_FILE_DIR:${target}>")
endforeach()
enable_testing()
add_test(NAME pdf_roundtrip COMMAND pdf_smoke)
add_test(NAME update_releases COMMAND update_smoke)
add_test(NAME annotation_ui COMMAND ui_smoke)
add_test(NAME pdf_corpus COMMAND corpus_smoke)
add_test(NAME language_spanish COMMAND language_smoke es)
add_test(NAME language_english COMMAND language_smoke en)
set_tests_properties(pdf_roundtrip annotation_ui pdf_corpus language_spanish language_english PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)

option(GPDF_BUILD_PERF_PROBE "Build opt-in profiling tool" OFF)
if(GPDF_BUILD_PERF_PROBE)
  add_executable(perf_probe tests/perf_probe.cpp src/PdfDocument.cpp src/Translations.cpp src/AnnotationExport.cpp src/SessionStore.cpp src/NativeScrollCursor.cpp src/CommentPopup.cpp resources/translations.qrc)
  target_link_libraries(perf_probe PRIVATE pdfium Qt6::Widgets Qt6::PrintSupport Qt6::Network)
  target_include_directories(perf_probe PRIVATE src)
  target_compile_definitions(perf_probe PRIVATE GPDF_PERF_PROBE)
  if(WIN32)
    target_link_libraries(perf_probe PRIVATE user32 gdi32)
  endif()
endif()

# Compare final widget pixels to a direct PDF raster at fractional and integer DPI.
foreach(scale 1 1.25 1.5 2)
  add_test(NAME render_quality_${scale} COMMAND render_quality "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/bookmarks.pdf")
  set_tests_properties(render_quality_${scale} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${scale}" TIMEOUT 90)
endforeach()
